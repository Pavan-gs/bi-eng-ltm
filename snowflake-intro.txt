-- ============================================================
-- 15. BASIC ANALYTICAL QUERY
-- ============================================================

-- Revenue by customer.

SELECT
    c.CUSTOMER_NAME,
    c.SEGMENT,
    SUM(f.SALES_AMOUNT) AS TOTAL_REVENUE
FROM DWH_TRAINING.DWH.FACT_SALES f
JOIN DWH_TRAINING.DWH.DIM_CUSTOMER c
    ON f.CUSTOMER_KEY = c.CUSTOMER_KEY
GROUP BY
    c.CUSTOMER_NAME,
    c.SEGMENT
ORDER BY
    TOTAL_REVENUE DESC;


-- ============================================================
-- 16. CREATE SIMPLE DATA MART
-- ============================================================

CREATE OR REPLACE TABLE DWH_TRAINING.MART.CUSTOMER_REVENUE
AS
SELECT
    c.CUSTOMER_NAME,
    c.SEGMENT,
    SUM(f.SALES_AMOUNT) AS TOTAL_REVENUE
FROM DWH_TRAINING.DWH.FACT_SALES f
JOIN DWH_TRAINING.DWH.DIM_CUSTOMER c
    ON f.CUSTOMER_KEY = c.CUSTOMER_KEY
GROUP BY
    c.CUSTOMER_NAME,
    c.SEGMENT;

1. The problem with simply INSERTing again

Suppose today's DWH contains:

C101 | Ravi  | Mumbai
C102 | Priya | Pune
C103 | Arun  | Delhi
C104 | Sneha | Mumbai

Tomorrow's source sends:

C101 | Ravi  | Bangalore
C102 | Priya | Pune
C105 | Meena | Chennai

There are three different situations:

C101 → existing customer + changed city
C102 → existing customer + no change
C105 → completely new customer

We therefore need logic:

If customer exists
       ↓
     UPDATE

If customer doesn't exist
       ↓
     INSERT

USE DATABASE DWH_TRAINING;

CREATE OR REPLACE TABLE DWH.DIM_CUSTOMER_INCREMENTAL
(
    CUSTOMER_ID   VARCHAR,
    CUSTOMER_NAME VARCHAR,
    CITY          VARCHAR,
    SEGMENT       VARCHAR
);

INSERT INTO DWH.DIM_CUSTOMER_INCREMENTAL
VALUES
('C101', 'Ravi',  'Mumbai', 'Retail'),
('C102', 'Priya', 'Pune',   'Premium'),
('C103', 'Arun',  'Delhi',  'Retail'),
('C104', 'Sneha', 'Mumbai', 'Premium');

SELECT *
FROM DWH.DIM_CUSTOMER_INCREMENTAL
ORDER BY CUSTOMER_ID;

# Let's simulate tomorrow's incoming data

CREATE OR REPLACE TEMPORARY TABLE STAGING.CUSTOMER_INCREMENTAL_LOAD
(
    CUSTOMER_ID   VARCHAR,
    CUSTOMER_NAME VARCHAR,
    CITY          VARCHAR,
    SEGMENT       VARCHAR
);

INSERT INTO STAGING.CUSTOMER_INCREMENTAL_LOAD
VALUES
('C101', 'Ravi',  'Bangalore', 'Retail'),
('C102', 'Priya', 'Pune',      'Premium'),
('C105', 'Meena', 'Chennai',   'Premium');


C101 → changed
C102 → unchanged
C105 → new

MERGE

MERGE INTO target
USING source
ON matching_condition

WHEN MATCHED THEN
    UPDATE ...

WHEN NOT MATCHED THEN
    INSERT ...;



             SOURCE
                |
                v
        Find matching key
           /         \
        MATCH       NO MATCH
          |             |
       UPDATE         INSERT

MERGE INTO DWH.DIM_CUSTOMER_INCREMENTAL AS target
USING STAGING.CUSTOMER_INCREMENTAL_LOAD AS source
ON target.CUSTOMER_ID = source.CUSTOMER_ID
WHEN MATCHED THEN
    UPDATE SET
        target.CUSTOMER_NAME = source.CUSTOMER_NAME,
        target.CITY          = source.CITY,
        target.SEGMENT       = source.SEGMENT

WHEN NOT MATCHED THEN
    INSERT
    (
        CUSTOMER_ID,
        CUSTOMER_NAME,
        CITY,
        SEGMENT
    )
    VALUES
    (
        source.CUSTOMER_ID,
        source.CUSTOMER_NAME,
        source.CITY,
        source.SEGMENT
    );

SELECT *
FROM DWH.DIM_CUSTOMER_INCREMENTAL
ORDER BY CUSTOMER_ID;
