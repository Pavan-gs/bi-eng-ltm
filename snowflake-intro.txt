-- ============================================================
-- SNOWFLAKE DWH TRAINING LAB
-- ============================================================


-- ============================================================
-- 1. CREATE COMPUTE
-- ============================================================

-- A virtual warehouse provides compute resources.
-- X-Small is sufficient for this training lab.
-- AUTO_SUSPEND helps avoid unnecessary compute usage.

CREATE OR REPLACE WAREHOUSE DWH_LAB_WH
WITH
    WAREHOUSE_SIZE = 'XSMALL'
    AUTO_SUSPEND = 60
    AUTO_RESUME = TRUE;

USE WAREHOUSE DWH_LAB_WH;


-- ============================================================
-- 2. CREATE DATABASE
-- ============================================================

-- A database is a high-level container for data objects.

CREATE OR REPLACE DATABASE DWH_TRAINING;

USE DATABASE DWH_TRAINING;


-- ============================================================
-- 3. CREATE SCHEMAS
-- ============================================================

-- RAW      = source-like data
-- STAGING  = cleaning/transformation
-- DWH      = dimensional warehouse
-- MART     = business-facing analytical data

CREATE OR REPLACE SCHEMA RAW;

CREATE OR REPLACE SCHEMA STAGING;

CREATE OR REPLACE SCHEMA DWH;

CREATE OR REPLACE SCHEMA MART;


-- ============================================================
-- 4. EXPLORE OBJECTS
-- ============================================================

SHOW DATABASES;

SHOW SCHEMAS IN DATABASE DWH_TRAINING;


-- ============================================================
-- 5. CREATE RAW TABLE
-- ============================================================

USE SCHEMA RAW;

CREATE OR REPLACE TABLE RAW_ORDERS
(
    ORDER_ID       VARCHAR,
    ORDER_DATE     DATE,
    CUSTOMER_ID    VARCHAR,
    REGION         VARCHAR,
    AMOUNT         NUMBER(12,2)
);

-- ============================================================
-- 8. CREATE STAGING TABLE
-- ============================================================

CREATE OR REPLACE TABLE DWH_TRAINING.STAGING.CLEAN_ORDERS
AS
SELECT
    ORDER_ID,
    ORDER_DATE,
    CUSTOMER_ID,
    UPPER(TRIM(REGION)) AS REGION,
    AMOUNT
FROM DWH_TRAINING.RAW.RAW_ORDERS;


-- ============================================================
-- 9. VIEW STAGING DATA
-- ============================================================

SELECT *
FROM DWH_TRAINING.STAGING.CLEAN_ORDERS;

-- 10. CREATE CUSTOMER DIMENSION

-- ============================================================
-- 12. CREATE FACT TABLE
-- ============================================================

-- IMPORTANT:
-- Grain:
-- One row represents ONE SALES TRANSACTION.

CREATE OR REPLACE TABLE DWH_TRAINING.DWH.FACT_SALES
(
    SALES_KEY     NUMBER,
    ORDER_ID      VARCHAR,
    CUSTOMER_KEY  NUMBER,
    ORDER_DATE    DATE,
    REGION        VARCHAR,
    QUANTITY      NUMBER,
    SALES_AMOUNT  NUMBER(12,2)
);

-- 14. EXPLORE DWH TABLES

-- ============================================================
-- 15. BASIC ANALYTICAL QUERY
-- ============================================================

-- Revenue by customer.

SELECT
    c.CUSTOMER_NAME,
    c.SEGMENT,
    SUM(f.SALES_AMOUNT) AS TOTAL_REVENUE
FROM DWH_TRAINING.DWH.FACT_SALES f
JOIN DWH_TRAINING.DWH.DIM_CUSTOMER c
    ON f.CUSTOMER_KEY = c.CUSTOMER_KEY
GROUP BY
    c.CUSTOMER_NAME,
    c.SEGMENT
ORDER BY
    TOTAL_REVENUE DESC;


-- ============================================================
-- 16. CREATE SIMPLE DATA MART
-- ============================================================

CREATE OR REPLACE TABLE DWH_TRAINING.MART.CUSTOMER_REVENUE
AS
SELECT
    c.CUSTOMER_NAME,
    c.SEGMENT,
    SUM(f.SALES_AMOUNT) AS TOTAL_REVENUE
FROM DWH_TRAINING.DWH.FACT_SALES f
JOIN DWH_TRAINING.DWH.DIM_CUSTOMER c
    ON f.CUSTOMER_KEY = c.CUSTOMER_KEY
GROUP BY
    c.CUSTOMER_NAME,
    c.SEGMENT;


-- ============================================================
-- 17. VIEW DATA MART
-- ============================================================

SELECT *
FROM DWH_TRAINING.MART.CUSTOMER_REVENUE;
